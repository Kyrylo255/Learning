SELECT — вибір стовпців для виведення
FROM — вибір таблиці
WHERE — фільтрація рядків за умовою
ORDER BY username ASC/DESC — сортування за зростанням/спаданням
; — завершення SQL-команди
LIKE - шукає текст за шаблоном
OR - окремі умови
IN (...)- список можливих значень
LIMIT 3 — повертає максимум 3 рядки.
IS NOT NULL — залишає рядки, де значення не є NULL
COUNT(*) — рахує всі рядки таблиці.
select DISTINCT role_name - прибирає повтори
GROUP BY role_name — групує користувачів за ролями.
HAVING — фільтрує групи після групування, наприклад за кількістю.
group by role_name HAVING count(role_name)> 1
